\title{\efxz{} Allocations with at Most Three Relevant Goods: Existence and a Verified Polynomial-Time Algorithm}
\titlerunning{\efxz{} with at Most Three Relevant Goods}

\author{Evan Lin}
\authorrunning{E. Lin}
\institute{University of Illinois Urbana-Champaign}

\maketitle

\begin{abstract}
An allocation of indivisible goods is \efxz{} if no agent envies another agent's bundle after removing any single good from it, even one the envious agent values at zero. We prove that every instance with nonnegative additive valuations in which each agent positively values at most three goods has a complete \efxz{} allocation, via a polynomial-time algorithm, K3ALG (for three relevant goods): a serial dictatorship whose leftover goods fill free slots, with the surplus going to one owner after at most one rotation of picks. For natural-number values, its correctness and a formally verified polynomial bound on the number of operations are machine-checked in Lean~4 on the program itself; the existence results are machine-checked over any linearly ordered cancellative commutative monoid, such as the nonnegative reals. If no agent values a good more than its other two together, some \efxz{} allocation gives more than two goods to at most one agent.
\keywords{Fair division \and Indivisible goods \and EFX \and Serial dictatorship \and Formal verification \and Lean}
\end{abstract}

\section{Introduction}

Envy-freeness up to any good is a central relaxation of envy-freeness for indivisible goods~\cite{AmanatidisBFV22}. Agents $N=[n]$ share goods $M=[m]$ with nonnegative additive valuations $v_i(S)=\sum_{g\in S}v_i(g)$. An allocation $X=(X_1,\dots,X_n)$ is \emph{complete} if it partitions $M$, and \emph{\efxz{}} if
\[
  v_i(X_i)\ \ge\ v_i(X_j\setminus\{h\})\qquad\text{for all } i\ne j \text{ and all } h\in X_j
\]
(vacuous if $X_j=\emptyset$); \emph{EFX} asks this only for $h$ with $v_i(h)>0$. So \efxz{} implies EFX, with equality for positive values. Completeness makes \efxz{} hard when agents value few goods: a good of $X_j$ worthless to $i$ forces $v_i(X_i)\ge v_i(X_j)$, so a good nobody values cannot go to an arbitrary agent, and one valued by $j$ but not by $i$ cannot join a bundle $i$ envies.

For additive valuations, complete EFX allocations are known only in special cases, mostly in preprints: three agents~\cite{ChaudhuryGM20}, $m\le n+3$~\cite{Mahara21}, four agents and at most nine goods~\cite{AlkassarFM26}, goods valued by at most two agents~\cite{AfshinmehrAMMS26,ChristodoulouFKS23,SgouritsaS25,KavianiSS24}, hypergraphs of girth at least four~\cite{LianeasSS26}. All use the all-goods form, \efxz{} (the graph results with zero values; for bounds on $n$ and $m$, positive values suffice by the perturbation below). For monotone valuations EFX can fail~\cite{AkramiMMSW26,MackenzieS26}. The graph results limit how many agents value a good; we limit how many goods an agent values: an instance is \emph{$k$-relevant} if every agent positively values at most $k$ goods. For $k\le2$ (a case of Theorem~\ref{thm:target}) serial dictatorship suffices, as formalized in~\cite{MrdEfx}. For $k\le4$, Viswanathan and Mehta~\cite{ViswanathanM24} announce ordinary EFX, with a proof sketch; their algorithm gives goods nobody values to an arbitrary agent, which can break \efxz{}. Such goods are easy to place (Lemma~\ref{lem:junk}), but their argument, written for EFX, does not address goods valued by one agent and worthless to another, which Section~\ref{sec:algo} handles.

A $3$-relevant counterexample would refute the additive EFX conjecture, as perturbing all values by a small $\varepsilon>0$ gives a positive instance without EFX allocations (compare~\cite[Lemma~1]{ChaudhuryGM20}); but this destroys the relevance structure, and no result we know of implies a positive answer~\cite{Repo}. We prove one, with a machine-checked polynomial-time algorithm; as \efxz{} implies EFX, this also proves the case $k=3$ of~\cite{ViswanathanM24} in full.

\begin{theorem}[existence]\label{thm:target}
Every instance with $n\ge1$ agents and nonnegative real additive valuations in which every agent positively values at most three goods has a complete \efxz{} allocation.
\end{theorem}

\begin{theorem}[one large bundle]\label{thm:D}
For $n\ge1$ agents with nonnegative real additive valuations, each valuing exactly three goods, none more than the other two together (\emph{balanced}), some complete \efxz{} allocation gives every agent but at most one at most two goods.
\end{theorem}

\noindent In Section~\ref{sec:correct}, a bundle of more than two goods has $m-2n+|\NA|+2$ goods (Proposition~\ref{prop:size}; written, not formalized). In the \emph{unit-cost model}, each list step, comparison of agents or goods, read of an input value or table entry, and comparison or addition of natural numbers of any size costs one operation.

\begin{theorem}[algorithm]\label{thm:algo}
Given $n\ge1$ agents, $m$ goods and natural-number values in binary, K3ALG (Section~\ref{sec:algo}) performs at most $400(n+m+1)^4$ operations in the unit-cost model, and if every agent positively values at most three goods, it returns a complete \efxz{} allocation.
\end{theorem}

\begin{corollary}[rational values; written]\label{cor:rat}
By scaling each agent's values to integers (which preserves every comparison), K3ALG applies to rational valuations; this corollary is not formalized.
\end{corollary}

\section{Preliminaries}\label{sec:prelim}

Let $R_i=\{g\st v_i(g)>0\}$. A bundle of at most one good is never strongly envied, and $i$ must not envy at all a bundle that contains a good outside $R_i$. If $|R_i|=3$ we write $R_i=\{a_i,b_i,c_i\}$ with $v_i(a_i)\ge v_i(b_i)\ge v_i(c_i)>0$, fix this order (ties broken by index) as $a_i\succ_i b_i\succ_i c_i$, and call $i$ \emph{balanced} if $v_i(a_i)\le v_i(b_i)+v_i(c_i)$, and \emph{strictly balanced} if the inequality is strict.

\begin{lemma}[peeling]\label{lem:peel}
Let $i\in N$ and $P\subseteq M$ satisfy (i)~$v_i(P)\ge v_i(M\setminus P)$ and (ii)~$|P|\le1$ or $P$ is worthless to every other agent. If the instance without $i$ and $P$ has an \efxz{} allocation $Y$, then $Y$ with $X_i=P$ is \efxz{}.
\end{lemma}

\noindent \emph{Rule R1} applies Lemma~\ref{lem:peel} with $P=\{p\}$ for a favourite remaining good $p$ of $i$ when $v_i(p)\ge v_i(M\setminus\{p\})$, and with $P=\emptyset$ when $v_i(p)=0$: then $i$ values every remaining good at zero, envies no bundle, and gets nothing, which constrains no one. If R1 fails for every agent, then $2v_i(g)\le2v_i(p)<v_i(M)$ for all $g$, so in a $3$-relevant instance every agent values exactly three goods and is strictly balanced.

\begin{lemma}[real values reduce to naturals; Appendix~\ref{app:prelim}]\label{lem:L12}
If $v$ is nonnegative real additive with $|R_i|\le3$ for all $i$, there are natural-number additive valuations $w$ with the same relevant goods such that $v_i(S)\le v_i(T)\iff w_i(S)\le w_i(T)$ for every agent $i$ and all $S,T\subseteq M$.
\end{lemma}

\section{Algorithm K3ALG}\label{sec:algo}

K3ALG has two stages: \emph{Stage R} peels agents by rule R1, and \emph{Stage L} runs construction \LBp{} on the remaining agents $A$ and goods $M$, each of which values exactly three goods and is strictly balanced.

\paragraph{Intuition.} Stage L starts with a serial dictatorship, Phase~1, that mimics R1: an agent with at most two goods left takes the better one; only if every waiting agent has all three does one, a \emph{leader}, take its top. Without leftover goods every bundle would be one pick, which nobody strongly envies. The difficulty is the leftovers: an agent that lost goods to earlier agents is safe only if those goods sit alone. For strictly balanced $i$ with distinct values this is exact (Lemma~\ref{lem:L5}, Appendix~\ref{app:neg}): holding only $b_i$, $i$ is safe iff $a_i$ is alone; only $c_i$, iff $a_i$ and $b_i$ are; $\{b_i,c_i\}$, always; $a_i$, unless $b_i,c_i$ share a bundle of three or more goods. \emph{Needs} record the goods that must sit alone; a \emph{frozen} agent holds one and takes nothing else; an \emph{upgrade} moves an agent from $b_i$ to $\{b_i,c_i\}$, releasing $a_i$ and using up a leftover; the one large bundle, the \emph{owner}'s, must respect the last case. The first candidate owner is the last agent $r$ not upgraded: nobody after $r$ needs its pick, and each agent that $r$'s bundle could threaten leads an earlier block, which supplies a slot for a protecting good.

\paragraph{Definitions} (dictionary and examples: Appendix~\ref{app:dict}). Let $G$ be the set of goods not yet picked (initially $M$). At its turn, agent $i$ takes its $\succ_i$-favourite good of $R_i\cap G$ as its \emph{pick} $Y_i$, or no pick ($Y_i=\none$). The next agent has at most two of its goods in $G$ if such an agent exists (an \emph{R1 step}), and otherwise finds all three available (an \emph{insertion step}); this is \emph{R1 priority}. A \emph{block} is an insertion step with the R1 steps after it; its inserted agent is its \emph{leader}. Step L3 then also gives $c_u$ to some agents $u$ whose pick is $b_u$; these \emph{upgraded} agents form $U$. The \emph{junk} $J$ is the set of goods still unallocated after Phase~1 and the upgrades. For $i\notin U$ the \emph{needs} are $N_i=\{g\in R_i\st g\succ_i Y_i\}$ ($N_i=R_i$ if $Y_i=\none$), and $\NA=\bigcup_{i\notin U}N_i$ is the set of goods \emph{needed alone}. An agent $i\notin U$ is \emph{frozen} if $Y_i\in\NA$, and a \emph{terminal} otherwise. The \emph{base} of $i$ is $\{Y_i\}$ (or $\emptyset$) for $i\notin U$ and $\{b_i,c_i\}$ for $i\in U$. A terminal may add junk up to two goods: its \emph{slots} are $\capa(i)=1$ or $2$ for a terminal with or without a pick, and $\capa(i)=0$ for other agents; $S=\sum_i\capa(i)$. A \emph{completion} gives every agent its base and each terminal $i$ at most $\capa(i)$ junk goods; \emph{without owner}, this places all junk, which needs $|J|\le S$; \emph{with owner} $o$ (a terminal or upgraded agent), $o$ also takes the rest of the junk. It satisfies the \emph{owner constraint} (OC) if no agent $x\notin U$, $x\ne o$, with $Y_x=a_x$ has $\{b_x,c_x\}\subseteq X_o$. The agents \emph{exposed} for $o$ are $E_o=\{x\notin U,\ x\ne o\st Y_x=a_x,\ \{b_x,c_x\}\subseteq J\cup\base(o)\}$; $\pi_x=\{b_x,c_x\}\cap J$. A \emph{need chain} from a frozen $x$ is a sequence $x=x_0,\dots,x_t$ of distinct agents in which each $x_{s+1}\notin U$ needs $Y_{x_s}$, $x_0,\dots,x_{t-1}$ are frozen, and $x_t$ is a terminal.

\medskip
\noindent\rule{\textwidth}{0.5pt}\par
\noindent\textbf{Algorithm K3ALG} (as formalized in \lean{EFX.K3.algoC}). Input: $n\ge1$ agents, $m$ goods, values $v_i(g)\in\mathbb{N}$; $A$ and $M$ start as all agents and goods. Lists are in index order; ``first'' means smallest index.
\par\noindent\textbf{P.} While $|A|\ge2$ and $M\ne\emptyset$: find the first $i\in A$ such that, for the first good $p\in M$ of largest $v_i(p)$, $v_i(p)=0$ or $v_i(M\setminus\{p\})\le v_i(p)$; if none, go to L1; else remove $i$ from $A$, and if $v_i(p)>0$, let $X_i=\{p\}$ and $M\leftarrow M\setminus\{p\}$. Then give $M$ to the agent of $A$, if $|A|=1$, and stop.
\par\noindent\textbf{L1.} For $i\in A$, sort $i$'s relevant goods in $M$ as $a_i,b_i,c_i$ (ties by index).
\par\noindent\textbf{L2.} \emph{Phase 1.} Repeatedly, the first unprocessed agent with at most two of its goods unpicked (else the first unprocessed agent) picks the first of $a_i,b_i,c_i$ still unpicked, if any.
\par\noindent\textbf{L3.} \emph{Upgrades.} $U\leftarrow\emptyset$. While some $k\notin U$ has $Y_k=b_k$, $c_k\in J$ and $b_k\notin\NA$ (for the current $U$), add the first such $k$ to $U$.
\par\noindent\textbf{L4.} If $|J|\le S$, return $\textsc{Complete}(\text{no owner},\emptyset)$.
\textbf{L5.} Let $r$ be the last-processed agent not in $U$ (Lemma~\ref{lem:r}), and $H=\textsc{HitSet}(E_r)$. If $|H|\le S-\capa(r)$, return $\textsc{Complete}(r,H)$.
\par\noindent\textbf{L6.} \emph{Rotation.} Let $k$ be the first agent of $E_r$ in $r$'s block. Scanning the agents processed after $k$ in order, append $j$ to the chain $k,\dots$ if the chain's last agent is frozen, $j\notin U$, and $j$ needs the pick of the chain's last agent. Each chain agent takes its predecessor's pick; $k$ takes the pick $b_k$ and the upgrade good $c_k$ and joins $U$.
\par\noindent\textbf{L7.} Recompute $J$, $\capa$ and $S$. If $|J|\le S$, return $\textsc{Complete}(\text{no owner},\emptyset)$; otherwise return $\textsc{Complete}(k,\textsc{HitSet}(E_k))$.\par
\noindent
$\textsc{HitSet}(E)$: if two agents $x\ne y$ of $E$ share a good of $\pi_x\cap\pi_y$ (the first such $x$, then $y$, then good among $b_x,c_x$), return that good followed by, for each other $z\in E$ in order, $b_z$ if $b_z\in J$ and $c_z$ otherwise; else return that list for all $z\in E$. As each $\pi_z\ne\emptyset$ (Theorems~\ref{thm:A} and~\ref{thm:B}), all entries are junk; there are at most $|E|-1$ if two sets meet, else $|E|$ (\lean{EFX.LB.hitSet_length}).
$\textsc{Complete}(o,H)$: picks go to their pickers, $c_u$ to each $u\in U$; the junk, $H$ first and then the rest of $J$, fills the slots of the agents other than $o$ in index order, each good at its first occurrence (so a repeat in $H$ is harmless); $o$ gets the remaining junk.\par
\noindent\rule{\textwidth}{0.5pt}\par

\section{Correctness}\label{sec:correct}

Stage R is correct by Lemma~\ref{lem:peel}, for any agent R1 applies to. For Stage L we prove more, for every Phase~1 order with R1 priority, upgrade order and need chain (construction \LBp{}), so K3ALG's smallest-index tie-breaking is not needed for correctness. Invariants of Phase~1 (Appendix~\ref{app:proofs}): (I1)~if $g\in R_i$ and $g\succ_i Y_i$ (or $Y_i=\none$), then $g$ was picked before $i$'s turn, and (B)~by an agent of $i$'s block; (I2)~if $g\in R_i$ is never picked, then $Y_i\succ_i g$; (I3)~an agent that finds all its goods available is a leader. R1 priority gives (B) and (I3), used in Theorem~\ref{thm:A}; the upgrades' stopping condition (UT) is used in Theorem~\ref{thm:B}. A \emph{pre-allocation} $(Y,U)$ is \emph{valid} if (V1)~$J\cap\NA=\emptyset$ and (V2)~$b_u,c_u\notin\NA$ for $u\in U$; after L3 it is (Lemma~\ref{lem:upgrades}).

\begin{theorem}[soundness]\label{thm:sound}
Let $(Y,U)$ be valid. Every completion satisfying (OC) is \efxz{} for every balanced additive valuation consistent with the rankings; only the owner's bundle can exceed two goods.
\end{theorem}

\begin{proof}[sketch]
A non-owner holds its base and at most $\capa(i)$ junk goods, two goods in all (a frozen one only its pick). So a bundle of two or more goods consists of a terminal's pick (not in $\NA$ by definition), goods $b_u,c_u$ (not in $\NA$ by (V2)) and junk (not in $\NA$ by (V1)): it has no good needed alone. Fix $i$ and such a bundle $X_j$. If $i\in U$, $X_j$ meets $R_i$ at most in $a_i$, and $v_i(a_i)\le v_i(b_i)+v_i(c_i)$ by balance. If $i\notin U$, the goods $i$ prefers to its pick are in $N_i\subseteq\NA$, so $X_j$ meets $R_i$ only below $Y_i$ (not at all if $Y_i=\none$). One such good is worth at most $v_i(Y_i)$; two exist only if $Y_i=a_i$ and $\{b_i,c_i\}\subseteq X_j$. If $|X_j|=2$, removing one leaves at most $v_i(a_i)$; if $|X_j|\ge3$, then $j=o$, and (OC) excludes exactly this (Appendix~\ref{app:proofs}).\qed
\end{proof}

\begin{lemma}[owner criterion]\label{lem:owner}
Let $(Y,U)$ be valid with $|J|>S$. If no $x\in E_o$ has $\{b_x,c_x\}\subseteq\base(o)$, and some $H\subseteq J$ with $|H|\le S-\capa(o)$ meets $\{b_x,c_x\}$ for every $x\in E_o$, then filling the other slots with $H$ first gives a completion with owner $o$ that satisfies (OC); $o$ is a \emph{valid owner}.
\end{lemma}

\begin{lemma}[the candidate $r$]\label{lem:r}
Some agent is not upgraded, and the last-processed such agent $r$ is a terminal: the first agent picks its top, so is not upgraded, and a $j\notin U$ needing $Y_r$ would come after $r$ by (I1) (Appendix~\ref{app:proofs}).
\end{lemma}

\begin{theorem}[the owner $r$]\label{thm:A}
Let $|J|>S$ after Phase 1 and the upgrades. Then $r$ is a valid owner unless the \emph{bad case} holds: an agent $k^*$ of $r$'s block is exposed for $r$ and frozen, every need chain from $k^*$ ends at $r$, and the sets $\pi_x$ ($x\in E_r$) are pairwise disjoint.
\end{theorem}

\begin{proof}[sketch]
Let $x\in E_r$. Each of $b_x,c_x$ is junk, so never picked, or $Y_r$, picked after $x$; so $x$ found all its goods available and is a leader (I3). So exposed agents lead distinct blocks, at most one, $k^*$, in $r$'s block, and $\pi_x\ne\emptyset$. The count $S-\capa(r)\ge|E_r|-1$ is an injection $x\mapsto\tau(x)$ from the exposed agents outside $r$'s block into the terminals other than $r$: $\tau(x)=x$ if $x$ is a terminal, and otherwise the end of a need chain from $x$, which exists and stays in $x$'s block, as by (B) an agent that needs a frozen agent's pick comes later in the same block. Distinct blocks give distinct terminals, each with $\capa\ge1$. One good of each $\pi_x$, one shared good if two sets meet, then fits in $S-\capa(r)$ unless $k^*$ exists, is frozen, has no need chain to another terminal, and the $\pi_x$ are disjoint (Appendix~\ref{app:proofs}).\qed
\end{proof}

\begin{theorem}[rotation]\label{thm:B}
In the bad case, rotate along any need chain $k^*=x_0,\dots,x_t=r$: each $x_s$ ($s\ge1$) takes $Y_{x_{s-1}}$ as its pick, and $k^*$ joins $U$ with pick $b_{k^*}$ and upgrade good $c_{k^*}$. The result is valid, and if its junk exceeds its slots, then $k^*$ is a valid owner.
\end{theorem}

\begin{proof}[sketch]
Chain agents get goods they needed, so $\NA$ shrinks, and $b_{k^*},c_{k^*}\in J\cup\{Y_r\}$ are free. Agents exposed for $k^*$ afterwards were exposed for $r$ ((UT) excludes $r$), disjointness keeps their pairs out of $\{b_{k^*},c_{k^*}\}$, and their terminals $\tau(x)$ lie outside $r$'s block and keep their slots (Appendix~\ref{app:proofs}; Example~\ref{ex:rot}).\qed
\end{proof}

\begin{theorem}[\LBp{} never fails]\label{thm:C}
If all agents are balanced, \LBp{} (Stage L with any Phase~1 order with R1 priority and upgrade order; owner $r$ if valid, else a rotation along any need chain from $k^*$ to $r$ and owner $k^*$ or none) returns a complete allocation, \efxz{} for every balanced additive valuation consistent with the rankings, with at most one bundle of more than two goods (by Lemmas~\ref{lem:upgrades},~\ref{lem:owner} and Theorems~\ref{thm:sound}--\ref{thm:B}).
\end{theorem}

\begin{proof}[of Theorems~\ref{thm:target}--\ref{thm:algo}: correctness]
If L5's test fails, $r$ is not a valid owner (Proposition~\ref{prop:O}), so the bad case holds (Theorem~\ref{thm:A}); L6's $k$ is $k^*$, its chain is a need chain from $k^*$, so ends at $r$, and Theorem~\ref{thm:B} applies to L7. So Stage L runs \LBp{} with R1 priority (Theorem~\ref{thm:C}, hence Theorem~\ref{thm:D}); with Lemma~\ref{lem:peel} and induction on $n$, K3ALG is correct; Lemma~\ref{lem:L12} extends existence to real values.\qed
\end{proof}

\section{Running Time and Verification}\label{sec:time}

\begin{proposition}[the owner test]\label{prop:O}
With $|J|>S$ after Phase~1 and the upgrades, $r$ is a valid owner if and only if $|E_r|\le S-\capa(r)$ or two of the sets $\pi_x$ ($x\in E_r$) meet; $\textup{\textsc{HitSet}}(E_r)$ is then a witness.
\end{proposition}

\noindent Indeed, the sets $\pi_x$ are nonempty with at most two goods each, such sets have a hitting set of fewer than $|E_r|$ goods iff two meet, and $S-\capa(r)\ge|E_r|-1$ (Theorem~\ref{thm:A}). In Lean, K3ALG is a program \lean{EFX.K3.algoC} in a \emph{cost monad} (a value with a count) whose value is \lean{EFX.K3.algo}, so Theorem~\ref{thm:algo} bounds the program that computes the allocation. The accounting belongs to the trusted base: traversals use charged combinators, and uncounted helpers sit under a \texttt{tick} bounding their work (Appendix~\ref{app:algo}). L3, which recomputes $\NA$ per candidate, dominates. Unit cost for numbers of any size is a strong assumption, removed by Corollary~\ref{cor:bits}.

\begin{corollary}[bit complexity; written, not formalized]\label{cor:bits}
With $b$ the largest bit length of an input value, K3ALG takes $O\big((n+m)^4(b+\log(n+m))\big)$ bit operations if each operation costs the bit length of its operands: it forms no products, and each number is a sum of at most $m$ values of one agent or a count at most $2(n+m)$ (Appendix~\ref{app:algo}).
\end{corollary}

\begin{corollary}[real values; written, not formalized]\label{cor:real}
With an oracle comparing sums of one agent's real values at unit cost, computing $w$ (Lemma~\ref{lem:L12}) and running K3ALG on it takes $O(nm)$ calls and $O((n+m)^4)$ operations (Appendix~\ref{app:algo}).
\end{corollary}

\paragraph{Verification.}\label{sec:lean} The formalization uses core Lean~4 (v4.34.0), no packages and no \texttt{sorry}; every declaration depends only on the axioms \texttt{propext}, \texttt{Classical.choice} and \texttt{Quot.sound}. The trusted base is the cost model and six definitions from~\cite{MrdEfx}; Theorem~\ref{thm:algo} is \lean{EFX.K3.algo_efx0} and \lean{EFX.K3.algoC_cost}; Theorems~\ref{thm:target} and~\ref{thm:D} are \lean{EFX.target_ordered} and \lean{EFX.corollaryD_ordered} over \lean{EFX.OrderedValue}, the axioms of a linearly ordered cancellative commutative monoid, such as $\mathbb{R}_{\ge0}$ (core Lean has no reals); Section~\ref{sec:correct} is formalized in full (Appendix~\ref{app:lean}). Allocations checked on all $6^n$ strict profiles of all connected cores with $n\le7$ verify Theorem~\ref{thm:target} for $n\le7$, as \efxz{} in cores is ordinal and ties need no check (Lemma~\ref{lem:L5}, Appendix~\ref{app:cert}).

\section{Conclusion}\label{sec:concl}

The case of four relevant goods per agent is open; there, the analogue of \LBp{} (Phase~1, upgrades, an owner or one rotation) fails on some runs already when one agent values four goods ($n=5$; checkers in~\cite{Repo}). The least size of the large bundle and the complexity of the owner test for arbitrary owners are open too.

\section{Proofs for Section~\ref{sec:prelim} and Cores}\label{app:prelim}

Full proofs of all reduction lemmas are in \texttt{proofs/lemmas.md} of~\cite{Repo}; we reproduce those used here.

\begin{proof}[of Lemma~\ref{lem:peel}]
Every other bundle lies in $M\setminus P$, so $i$ values it at most $v_i(M\setminus P)\le v_i(P)=v_i(X_i)$. Towards $P$, every other agent sees at most one good or a bundle worthless to it (and nothing if $P=\emptyset$). The remaining constraints are those of $Y$.\qed
\end{proof}

\begin{lemma}[unvalued goods]\label{lem:junk}
Let $Z\ne\emptyset$ be the set of goods valued by no agent. If the instance without $Z$ has an \efxz{} allocation, so does the instance.
\end{lemma}

\begin{proof}
Let $Y$ be \efxz{} for $M\setminus Z$. While the envy graph ($i\to j$ iff $v_i(Y_j)>v_i(Y_i)$) has a cycle, give each agent on it the bundle of its successor. The bundles are the same and no agent loses value; an agent that now sees its old bundle satisfies $v_i(Y_i\setminus\{h\})\le v_i(Y_i)$, so \efxz{} is preserved. The sum of values strictly increases, so this stops, with an acyclic envy graph. Let $s$ be a source: $v_i(Y_s)\le v_i(Y_i)$ for all $i$. Put $X_s=Y_s\cup Z$. For $i\ne s$, $v_i(X_s\setminus\{h\})\le v_i(X_s)=v_i(Y_s)\le v_i(Y_i)$; all other constraints are unchanged.\qed
\end{proof}

\paragraph{Cores.} Agent $i$ is \emph{top-heavy} if $v_i(a_i)\ge v_i(b_i)+v_i(c_i)$. \emph{Rule R2} gives $i$ the set $P$ of its relevant goods that no other agent values, if $|P|\ge2$ and (i) of Lemma~\ref{lem:peel} holds. A \emph{core} is an instance with at least two agents, each valuing exactly three goods and strictly balanced, each with at most one \emph{private} good (one no other agent values), and no good valued by nobody. Cores are the instances that the reductions below cannot simplify; the exhaustive certification (Appendix~\ref{app:cert}) and the counterexamples (Appendix~\ref{app:neg}) are about cores.

\begin{proposition}[core reduction]\label{prop:core}
If every core with at most $N_0$ agents has an \efxz{} allocation, so does every $3$-relevant instance with between $1$ and $N_0$ agents.
\end{proposition}

\begin{proof}[of Proposition~\ref{prop:core}]
Induction on $n+m$ over $3$-relevant instances with $1\le n\le N_0$. One agent takes everything. Otherwise: (1)~if some good is valued by nobody, use Lemma~\ref{lem:junk}; (2)~else if some agent has at most two relevant goods or is top-heavy, use rule R1 and Lemma~\ref{lem:peel}; (3)~else if some agent $i$ has at least two private goods, let $P$ be all of them: if $|P|=3$, (i) is trivial, and if $P=\{x,y\}$ and $R_i\setminus P=\{z\}$, then $v_i(x)+v_i(y)\ge v_i(b_i)+v_i(c_i)>v_i(a_i)\ge v_i(z)$, so Lemma~\ref{lem:peel} applies (rule R2); (4)~else the instance is a core, covered by hypothesis. Each case passes to a smaller $3$-relevant instance.\qed
\end{proof}

\noindent In Lean this is \lean{EFX.core_reduction} (with \lean{EFX.peel}, \lean{EFX.peelEmpty}, \lean{EFX.peelR2} and \lean{EFX.Inst.exists_efx0_of_junk}).

\begin{proof}[of Lemma~\ref{lem:L12}]
Fix $i$; $w_i$ is built from $v_i$ alone. Goods outside $R_i$ contribute $0$ to both sides under both valuations, and removing goods common to $S$ and $T$ changes neither comparison, so it suffices that $w_i$ preserves $v_i(X)\le v_i(Y)$ for disjoint $X,Y\subseteq R_i$. Name the relevant goods with values $a\ge b\ge c>0$. Comparisons with an empty side are decided by positivity; two single goods by the ties among $a,b,c$; a single good against the other two by the sign of $a-(b+c)$, because $b<a+c$ and $c<a+b$ always hold. Two disjoint pairs do not fit in three goods. Choose $w_i$ on $R_i$ (in the same order) by the pattern: $(4,3,2)$ if $a>b>c$ and $a<b+c$; $(3,2,1)$ if $a>b>c$ and $a=b+c$; $(5,2,1)$ if $a>b>c$ and $a>b+c$; $(2,2,1)$ if $a=b>c$; $(3,2,2)$, $(2,1,1)$, $(3,1,1)$ if $a>b=c$ and $a<2b$, $a=2b$, $a>2b$; $(1,1,1)$ if $a=b=c$; $(2,1)$ or $(1,1)$ for two relevant goods; $(1)$ for one; and $w_i(g)=0$ for $g\notin R_i$. Each triple has the same ties and the same sign of $a-(b+c)$ as its pattern.\qed
\end{proof}

\noindent Lean: \lean{EFX.l12}, which uses all 31 permutations of these representatives (checked by \texttt{decide}) instead of sorting, and \lean{EFX.efx0_iff_of_agree} for the transfer of \efxz{}. The choice of pattern needs at most three comparisons to sort $a,b,c$, two for the ties and one of $a$ with $b+c$, as used in Corollary~\ref{cor:real}.

\section{Proofs for Section~\ref{sec:correct}}\label{app:proofs}

These are the proofs of \texttt{proofs/lb\_last\_step.md} of~\cite{Repo}, Sections 1--6. A pair $P=(Y,U)$ as in Section~\ref{sec:algo} is a \emph{pre-allocation}; $F$ and $T$ denote its frozen agents and its terminals, and $\omega=|J|-S$. After the rotation, primes denote the new state: $P'=(Y',U')$, $J'$, $\NA'$, $\capa'$, $S'$, $\omega'$, $E'_{k}$.

\paragraph{Phase 1.} (I1): $i$ took its favourite remaining good, and goods leave $G$ only by being picked. (I2): $g$ was still in $G$ at $i$'s turn. (I3): R1 steps process only agents with at most two goods left. (B): each block starts with an insertion step, when every unprocessed agent $i$ has $R_i\subseteq G$ (by R1 priority); by (I1) $g$ was picked before $j$'s turn, and not in an earlier block, since $j$ was unprocessed when its own block started.

\begin{proof}[of Theorem~\ref{thm:sound}]
Only $X_o$ can exceed two goods. Fix $i$ and $j\ne i$ with $|X_j|\ge2$. Then $j$ is not frozen, so $X_j\cap\NA=\emptyset$ by (V1) and (V2). If $i\in U$, then $X_j$ meets $R_i$ in at most $a_i$, and $v_i(X_j\setminus\{h\})\le v_i(a_i)\le v_i(b_i)+v_i(c_i)\le v_i(X_i)$. If $Y_i=\none$, then $R_i=N_i\subseteq\NA$ and $X_j$ is worthless to $i$. Otherwise every good of $R_i\cap X_j$ is outside $N_i\cup\{Y_i\}$, hence below $Y_i$. If there is at most one, $v_i(X_j\setminus\{h\})\le v_i(Y_i)\le v_i(X_i)$. If there are two, then $Y_i=a_i$ and $R_i\cap X_j=\{b_i,c_i\}$; for $|X_j|=2$ removing one leaves a good worth at most $v_i(a_i)$, and $|X_j|\ge3$ means $j=o$, excluded by (OC).\qed
\end{proof}

\begin{proof}[of Lemma~\ref{lem:owner}]
Since $|J|=S+\omega>S-\capa(o)$, $H$ extends to $C\subseteq J$ with $|C|=S-\capa(o)$, which exactly fills the slots of the terminals other than $o$; let $X_o=\base(o)\cup(J\setminus C)$. If $x\notin U$, $x\ne o$, $Y_x=a_x$ had $\{b_x,c_x\}\subseteq X_o$, then $\{b_x,c_x\}\subseteq J\cup\base(o)$, so $x\in E_o$, and $H\subseteq C$ meets $\{b_x,c_x\}$ outside $X_o$, a contradiction.\qed
\end{proof}

\begin{lemma}[upgrades]\label{lem:upgrades}
When the upgrade loop stops, $(Y,U)$ is a valid pre-allocation, and (UT) no $k\notin U$ has $Y_k=b_k$, $c_k\in J$ and $b_k\notin\NA$.
\end{lemma}

\begin{proof}
(UT) is the stopping condition. (V1): by (I2) a junk good lies below the pick of every agent that values it, so it is in no $N_i$. (V2): $c_u$ was junk; $b_u\notin\NA$ held when $u$ was upgraded, and $\NA$ only shrinks as $U$ grows.\qed
\end{proof}

\begin{proof}[of Lemma~\ref{lem:r} and Theorem~\ref{thm:A}]
(A0)~$r$ exists: the first processed agent is inserted and holds its top, so it is not upgraded; in Lean, \lean{EFX.LB.lastOut_exists}.
(A1)~$r\in T$: if $Y_r=\none$, $r$ is not frozen by definition; if $Y_r\in N_j$ for some $j\notin U$, then by (I1) $j$ was processed after $r$, contradicting the choice of $r$ (\lean{EFX.LB.lastOut_terminal}).
(A2)~Every agent processed after $r$ is in $U$, by the choice of $r$.
(A3)~Let $x\in E_r$. Then $x$ was processed before $r$ (A2). At $x$'s turn, $a_x=Y_x$, every junk good, and $Y_r$ were available, and $\{b_x,c_x\}\subseteq J\cup\{Y_r\}$, so $R_x\subseteq G$ and $x$ is a leader by (I3). Hence each block contains at most one exposed agent; let $B^*$ be $r$'s block and $k^*$ its exposed agent, if any.
(A4)~For a frozen $x$, some $j\notin U$ has $Y_x\in N_j$, and by (B) $j$ is processed after $x$ in $x$'s block. Iterating while the current agent is frozen gives a need chain of distinct agents of $x$'s block ending at a terminal. Let $\tau(x)=x$ if $x\in T$, and otherwise the end of some need chain from $x$; it has $\capa\ge1$.
For $x\in E_r$, $\pi_x=\{b_x,c_x\}\cap J\ne\emptyset$, since at most one of $b_x,c_x$ is $Y_r$; for the same reason no exposed pair lies in $\base(r)$, the first condition of Lemma~\ref{lem:owner}.

The agents $\tau(x)$, $x\in E_r\setminus B^*$, lie in distinct blocks other than $B^*$, so they are distinct terminals other than $r$, and $S-\capa(r)\ge|E_r\setminus B^*|$. Let $H$ contain one good of each $\pi_x$, a common one when two meet. If $E_r\cap B^*=\emptyset$, then $|H|\le|E_r|\le S-\capa(r)$. Otherwise $E_r\cap B^*=\{k^*\}$, $k^*\ne r$ and $|E_r\setminus B^*|=|E_r|-1$. If $k^*\in T$, or some need chain from $k^*$ ends at a terminal other than $r$, then $\tau(k^*)$ can be chosen in $B^*\setminus\{r\}$, one more terminal, so $S-\capa(r)\ge|E_r|\ge|H|$. If two sets $\pi_x$ meet, then $|H|\le|E_r|-1\le S-\capa(r)$. If none of these holds, the bad case holds.\qed
\end{proof}

\begin{proof}[of Theorem~\ref{thm:B}]
Let $k=k^*$, $W=J\cup\{Y_r\}$; the new junk is $J'=W\setminus\{b_k,c_k\}$, and every pick other than $Y_r$ is still a pick in $P'$. In $P'$, $k\in U'$ with $Y'_k=b_k$ and upgrade good $c_k$, so $Y'_u=b_u$ holds for every $u\in U'$.
(a)~$P'$ is a pre-allocation: $Y'_{x_s}\in R_{x_s}$ since $x_s$ needed it, and $b_k,c_k\in W$ are neither picks of $P'$ other than $Y'_k$ nor goods $c_u$ with $u\in U$.
(b)~$\NA'\subseteq\NA$: $Y'_{x_s}\succ Y_{x_s}$ (or $Y_{x_s}=\none$), so $N'_{x_s}\subseteq N_{x_s}$; $k$ had $N_k=\emptyset$ and is now in $U'$.
(c)~Validity: $J'\subseteq W$, $J\cap\NA=\emptyset$ and $Y_r\notin\NA$ (A1), so $J'\cap\NA'=\emptyset$; for $u\in U$, $b_u,c_u\notin\NA\supseteq\NA'$; $b_k,c_k\in W$ are not in $\NA$.
(d)~A terminal $x\notin B^*$ keeps its pick and is a terminal of $P'$ with the same slots.
(e)~$E'_k\subseteq E_r\setminus\{k\}$, where $E'_k$ is taken in $P'$, with $J'\cup\{b_k,c_k\}\subseteq W$. An agent off the chain in $E'_k$ is in $E_r$. An interior chain agent $x_s$ ($1\le s<t$) is not: $a_{x_s}=Y'_{x_s}\succ Y_{x_s}$, so $Y_{x_s}\in\{b_{x_s},c_{x_s}\}$, and $Y_{x_s}\notin W$. Nor is $r$: $Y_r\ne a_r$, since $r$ needs $Y_{x_{t-1}}$. If $Y_r=\none$ or $Y_r=c_r$, then $b_r$ was picked by another agent (I1), so $b_r\notin W$. If $Y_r=b_r$, then (UT) applies to $r$ ($b_r\notin\NA$ by (A1)), so $c_r\notin J$, and $c_r\ne Y_r$, so $c_r\notin W$.
(f)~If some $x\in E'_k$ had $\{b_x,c_x\}=\{b_k,c_k\}$, then $x\in E_r\setminus\{k\}$ and $\pi_x=\pi_k\ne\emptyset$, against disjointness. So every $x\in E'_k$ has a good of $\{b_x,c_x\}$ in $J'$: $\textsc{HitSet}(E'_k)$ in L7 lists junk goods.
(g)~Let $H'$ contain one good of $\{b_x,c_x\}\cap J'$ for each $x\in E'_k$. By (e), $|H'|\le|E_r|-1=|E_r\setminus B^*|$. The terminals $\tau(x)$, $x\in E_r\setminus B^*$, are distinct, outside $B^*$, and keep $\capa'\ge1$ by (d). So $|H'|\le S'=S'-\capa'(k)$, and Lemma~\ref{lem:owner} applies to $P'$ and $o=k$.\qed
\end{proof}

\begin{proposition}[size of the large bundle; written, not formalized]\label{prop:size}
Let $(Y,U)$ be valid with $\omega=|J|-S\ge1$. Then $\omega=m-2n+|\NA|$ (with $n$, $m$ the agents and goods of Stage L), and every completion with owner $o$ in which the other terminals' slots are full gives $o$ exactly $\omega+2$ goods; so do the completions of Lemma~\ref{lem:owner} and of K3ALG.
\end{proposition}

\begin{proof}
In a valid pre-allocation each good of $\NA$ is the pick of an agent outside $U$ (not junk by (V1), not $b_u,c_u$ by (V2)), which is then frozen, and each frozen agent's pick is in $\NA$; so $|F|=|\NA|$. Let $T_1$ and $T_0$ be the terminals with and without a pick. Then $m=|U|+|F|+|T_1|+|U|+|J|$ and $S=|T_1|+2|T_0|$, so $\omega=m-2(|U|+|F|+|T_1|+|T_0|)+|F|=m-2n+|\NA|$. The owner gets its base and $|J|-(S-\capa(o))=\omega+\capa(o)$ junk goods, and $|\base(o)|+\capa(o)=2$ for every terminal or upgraded agent.\qed
\end{proof}

\noindent The rotation does not increase $\NA$ (step (b) of the proof of Theorem~\ref{thm:B}), so it does not increase the large bundle.

\section{Running Time: Cost Table and Measurements}\label{app:algo}

The program is the listing of Section~\ref{sec:algo}, formalized as \lean{EFX.K3.algoC}; see \texttt{proofs/k3\_algorithm.md} of~\cite{Repo}. In L5 and L6 the program completes with owner $r$ (or without owner) if $k$ (or $r$) does not exist; by Lemma~\ref{lem:r} and Theorem~\ref{thm:A} this never happens.

\paragraph{Cost table (Lean bound with $N=n+m+1$; finer bound written, not formalized).} Stage R: $15N^2$ per round, $O(n^2m)$ in all. Rankings: $14N^2$. Processing order: $15N^3$. Phase~1's picks: $17N^3$. Upgrades: $36N^4$, from at most $n$ rounds, each testing $n$ candidates, each recomputing whether $b_k\in\NA$ by scanning $n$ agents with a membership test in $U$; $O(n^4+n^2m)$ in the finer count, the dominant term. Junk and slots: $O(N^3)$. Owner test: $20N^2+24N^3$. Blocks, $k^*$, chain, rotation, completion: $O(N^3)$. Output: $4N^2$. The total is at most $400N^4$ (\lean{EFX.K3.algoC_cost}), also at most $6400(n+m)^4$. Summing the finer column gives $O(n^4+n^2m)$ operations (written, not formalized). On random instances the count is far below the bound (ratio $0.002$ at $n=30$).

\paragraph{The accounting discipline.} The cost monad \lean{EFX.Timed} (\texttt{Timed.lean}) pairs a value with a count; \texttt{tick~$k$} adds $k$, and sequencing adds the counts of the steps actually evaluated. The program keeps to four rules. (1)~Every list traversal goes through a charged combinator (\lean{mapC}, \lean{filterC}, \lean{anyC}, \lean{findC}, \lean{memC}, \lean{eraseC}, \lean{sumMapC}, \lean{lengthC}, \dots), one unit per cell visited plus the cost of the element function; the recursions of Phase~1, the blocks, the chain, the rotation and the completion charge one unit per step. (2)~Every call of an uncounted function sits under a \lean{tick} that bounds its work: \lean{sort3C} is \texttt{tick 10} (at most four list cells, three value reads, three comparisons), \lean{rankC} is \texttt{tick 6} (three table reads, three comparisons), \lean{finRangeC k} is \texttt{tick~$k$}. (3)~No recursion over lists runs outside the monad; each counted definition has a value lemma saying that it computes the uncounted one, so every loop of the specification is a counted loop. (4)~Tables (rankings, picks, blocks, slots, the output) are arrays filled once by \lean{mkTable}, at the cost of their entries plus one per entry, and each read costs one. Abstracted, as disclosed in the files: arithmetic on natural numbers of any size (one unit), reading $v_i(g)$ (one unit), and building pairs, options and list cells and matching on them. An auditor should read each \lean{tick} and each \lean{pure} in \texttt{Timed.lean}, \texttt{K3CostLB.lean} and \texttt{K3Cost.lean} and check that it covers the work it stands for; the value lemmas and the bound are then machine-checked.

\begin{proof}[of Corollary~\ref{cor:bits}]
We read the counted program (\texttt{K3Cost.lean}, \texttt{K3CostLB.lean}, \texttt{Timed.lean}); it forms no products. Values enter only in Stage~R and in L1: each is compared with $0$ or with another value of the same agent, and R1 compares $v_i(p)$ with $v_i(M\setminus\{p\})$, formed by adding values of agent $i$ over the remaining goods. So every number derived from values is a sum of at most $m$ values of one agent, below $m2^b$, and Stage~L reads no values. The other numbers are ranks in $\{0,\dots,3\}$, slot counts and their sums (at most $2n$), list lengths (at most $n+m$) and block counters (at most $n$), and agents and goods are indices below $n$ and $m$. So every operand, including table addresses, has $O(b+\log(n+m))$ bits, and each unit operation costs that many bit operations; multiply by Theorem~\ref{thm:algo}'s bound.\qed
\end{proof}

\begin{proof}[of Corollary~\ref{cor:real}]
Lemma~\ref{lem:L12} is constructive: finding $R_i$ takes $m$ comparisons with $0$, and at most six more comparisons (Appendix~\ref{app:prelim}) give the ties among $a_i,b_i,c_i$ and the sign of $v_i(a_i)-v_i(b_i)-v_i(c_i)$, which select $w_i$, with entries at most $5$. K3ALG on $w$ takes $O((n+m)^4)$ operations (Theorem~\ref{thm:algo}), and its output is \efxz{} for $w$, hence for $v$ (Lemma~\ref{lem:L12}, \lean{EFX.efx0_iff_of_agree}).\qed
\end{proof}

\paragraph{A faster implementation (empirical).} A Python implementation of the same algorithm replaces the naive scans by heaps (for Stage R and the processing order), per-good counters of the agents that rank a good above their pick (for the upgrades) and an index from junk goods to exposed agents (for \textsc{HitSet}). Its written analysis, not formalized, gives $O((n+m)\log n)$ operations after reading the input. It agrees with a literal transcription of the Lean definitions on $200{,}000$ random instances and, with Lean's own evaluation of \lean{EFX.K3.algo}, on $675$ instances; every output passes a raw \efxz{} check, and every Lean operation count is below the proved bound. It takes $1.53$\,s and $1.58$\,s (medians) on random cores with $10^5$ agents and $1.5\times10^5$ and $2\times10^5$ goods. These are measurements, not guarantees.

\section{Ordinality of Cores and the Counterexamples}\label{app:neg}

A good is \emph{alone} if it is the only good of its bundle.

\begin{lemma}[cases of safety]\label{lem:L5}
Let agent $i$ have $R_i=\{a,b,c\}$ with $v_i(a)>v_i(b)>v_i(c)>0$ and $v_i(a)<v_i(b)+v_i(c)$. Then $i$ satisfies the \efxz{} constraints iff one of: (T)~$i$ holds $a$, and either $i$ holds $b$ or $c$, or $b$ and $c$ are not together in a bundle of at least three goods; (P)~$i$ holds $b$ and $c$; (B)~$i$ holds $b$, and $a$ is alone; (C)~$i$ holds $c$, and $a$ and $b$ are alone; (E)~$a$, $b$ and $c$ are all alone.
\end{lemma}

\begin{proof}
The subset sums are ordered $0<c<b<a<b+c<a+c<a+b<a+b+c$ (writing $x$ for $v_i(x)$), so every constraint is decided by the ranking. Consider the goods of $R_i$ that $i$ holds. If it holds $a$ and one more, every other bundle meets $R_i$ in one good, worth $\le b<a+c$. If it holds $\{b,c\}$, the only threat is $a<b+c$. If it holds only $a$: bundles meeting $R_i$ in one good are harmless; $\{b,c\}$ as a bundle threatens $b<a$; $b,c$ together with another good threaten $b+c>a$. If it holds only $b$: $a$ not alone lies with $c$ (threat $a$) or with a good outside $R_i$ (threat $\ge a$), both $>b$; if $a$ is alone, $c$'s bundle threatens at most $c<b$. If it holds only $c$, it needs $a$ and $b$ alone by the same argument. If it holds none of $a,b,c$, any bundle with a good of $R_i$ and at least two goods threatens a positive amount, so all three must be alone.\qed
\end{proof}

\noindent The cases remain sufficient with ties, so tied cores reduce to strict ranking profiles (\texttt{proofs/lemmas.md}, L5).

\begin{proposition}[limits of the shape]\label{prop:neg}
Take strict rankings and strictly balanced values.
(a) Let agents $0,1,2$ rank $g_0\succ g_1\succ p_i$, where $p_i$ is private to agent $i$. No \efxz{} allocation has all bundles of at most two goods, while $\{g_0\},\{g_1\},\{p_0,p_1,p_2\}$ is \efxz{}.
(b) Let agents $0,1,2,3$ rank $g_0\succ g_2\succ p_0$, $g_0\succ g_2\succ p_1$, $g_1\succ g_2\succ p_2$, $g_1\succ g_2\succ p_3$. Every \efxz{} allocation with at most one bundle of more than two goods has bundle sizes $4,1,1,1$.
\end{proposition}

\noindent Exhaustive computation shows that no core with two agents needs a bundle of more than two goods, so (a) is a smallest example; see \lean{proofs/counterexamples.md} and Section~1 of \lean{proofs/construction.md} in~\cite{Repo}.

\begin{proof}[of Proposition~\ref{prop:neg}]
(a)~With five goods in three bundles of at most two goods, the sizes are $2,2,1$, so exactly one good is alone. Good $g_0$ is everyone's top and has one holder, so two agents do not hold their top. For them (T) fails, (C) and (E) need two or three alone goods, so each is in case (P) or (B), and both hold $g_1$: impossible. In $\{g_0\},\{g_1\},\{p_0,p_1,p_2\}$, the holder of $g_0$ is in case (T), the holder of $g_1$ in case (B), and the third agent in case (C).
(b)~Each of the tops $g_0,g_1$ has a claimant that does not hold it (a \emph{loser}). A loser is safe only by (P) or (B), which need it to hold $g_2$, or by (C) or (E), which need $g_2$ alone. The two losers are distinct and cannot both hold $g_2$, so $g_2$ is alone, and no agent is in case (P). A loser holding $\{g_2\}$ is in case (B) or (E) and needs its top alone; a loser not holding $g_2$ is in case (C) or (E) and needs its top and $g_2$ alone. So $g_0,g_1,g_2$ are alone, held by three agents, and the fourth bundle is $\{p_0,p_1,p_2,p_3\}$. The allocation giving $g_0$ to agent~0, $g_1$ to agent~2, $g_2$ to agent~1 and the private goods to agent~3 is \efxz{} (cases T, B, T, C).\qed
\end{proof}
